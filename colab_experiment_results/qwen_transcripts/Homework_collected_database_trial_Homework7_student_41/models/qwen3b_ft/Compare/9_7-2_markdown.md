a) \( V_t = AV \)

\[
\frac{1}{LC}(i) = \frac{A}{LC} + 0 \quad \text{or} \quad A = 2
\]

\[
V_e = 2V
\]

b) \( V_s = 0.2t \)

\[
V_f = AR + B \quad \Rightarrow \quad \frac{dV}{dt} = A
\]

\[
\frac{1}{LC}(0.2t) = \frac{4R + B}{LC} + \frac{AR}{L} = 7
\]

\[
\frac{0.2V}{t} = \frac{4R}{LC} \quad \Rightarrow \quad A = 0.2
\]

\[
\therefore B = -0.29c = -0.0012
\]

\[
V_e = 0.2t - 0.0012V
\]

c) \( V_s = e^{-30t} \)

\[
V_f = Ae^{-20t} \quad \Rightarrow \quad \frac{dV}{dt} = -30Ae^{-20t} \quad \Rightarrow \quad \frac{d^2V}{dt^2} = 900Ae^{-20t}
\]

\[
\frac{e^{-20t}}{LC} = \frac{Ae^{-20t}}{LC} + \frac{30RAe^{-20t}}{L} + \frac{900Ae^{-20t}}{L}
\]

\[
12000e^{-30t} = 900Ae^{-20t} + 2100Ae^{-20t} + 12000Ae^{-30t}
\]

\[
10800Ae^{-20t} = 12000e^{-30t}
\]

\[
\therefore A = \frac{10}{9}
\]

\[
V_f = \frac{10}{9}e^{-30t}V
\]
