\[
V_C(+) = 5\sqrt{2}\cos(8t) + 2\sin(8t + 45^\circ)
\]
\[
= 5\sqrt{2}\cos(8t) + 2\cos(8t - 45^\circ)
\]
\[
= 5\sqrt{2}\angle 0^\circ + 2\angle 45^\circ
\]
\[
= \textcolor{red}{5\sqrt{2}} + \sqrt{2} - \sqrt{2}
\]
\[
= \textcolor{red}{6\sqrt{2}} - \sqrt{2}
\]
\[
\Rightarrow \sqrt{(\textcolor{red}{6\sqrt{2}})^2 + (\sqrt{2})^2} = \sqrt{74}
\]
\[
\tan^{-1}\left(\frac{-\sqrt{2}}{\textcolor{red}{6\sqrt{2}}}\right) = -9.46^\circ
\]
\[
V_C(+) = \sqrt{74}\cos(8t - 9.46^\circ) \quad \textcolor{red}{V}
\]
