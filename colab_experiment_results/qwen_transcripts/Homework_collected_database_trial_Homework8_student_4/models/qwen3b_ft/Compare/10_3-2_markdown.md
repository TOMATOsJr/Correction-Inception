\[
v(t) = 5\sqrt{2}\cos(8t) + 2\sin(8t + 45^\circ) \quad V
\]
\[
= 5\sqrt{2}\cos(8t) + 2\cos(8t - 45^\circ)
\]
\[
= 5\sqrt{2}e^{j0^\circ} + 2e^{-j45^\circ} = 5\sqrt{2} + 2\left(\frac{\sqrt{2}}{2} - \frac{\sqrt{2}}{2}j\right) = 6\sqrt{2} - \sqrt{2}j
\]

\[
r = \sqrt{(6\sqrt{2})^2 + (\sqrt{2})^2} = \textcolor{red}{\sqrt{78}}, \quad \phi = \tan^{-1}\left(\frac{\sqrt{2}}{6\sqrt{2}}\right) = -9.462^\circ
\]

\[
\Rightarrow \sqrt{\frac{78}{\textcolor{red}{\sqrt{78}}}}e^{-j9.462^\circ}
\]

\[
\Rightarrow v(t) = \frac{6\sqrt{2}}{\sqrt{78}}\cos(8t - 9.462^\circ) \quad V
\]
