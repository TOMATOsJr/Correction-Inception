\[
i_2 + 2 = 3 \quad \Rightarrow \quad i_2 = 1 \text{ mA} = 0.001 \text{ A}
\]

\[
3 = 5 + i_3 \quad \Rightarrow \quad i_3 = -2 \text{ mA} = -0.002 \text{ A}
\]

\[
P = 3(-0.002) = -0.006 \text{ W}
\]

\[
P = 2(0.001) = 0.002 \text{ W}
