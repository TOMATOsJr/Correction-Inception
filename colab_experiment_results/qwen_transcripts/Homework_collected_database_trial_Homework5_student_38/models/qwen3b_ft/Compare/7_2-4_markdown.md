\[

\[
v(t_0) = -1 \, \text{mV}
\]

\[
v(t) = \frac{i}{C} \cdot t = \frac{4 \cdot 10^{-6} \, A}{2 \cdot 10^{-12} \, F} \cdot 10^{-9} \, s = 2 \, \text{mV/s}
\]

v(\text{mV}) = \begin{cases}

\end{cases}
